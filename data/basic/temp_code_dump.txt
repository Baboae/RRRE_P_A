    user_mp_info = fetch_json(f"https://game.raceroom.com/multiplayer-rating/user/{userid}.json")
    if user_mp_info is None:

    else:
      save_json(user_mp_info, "user_mp_info.json", user_folder)

      RacesCompleted = user_mp_info["RacesCompleted"]
      NumberOfPages = math.ceil(RacesCompleted/100)

      for i in range(1,NumberOfPages+1):
        CareerPageURL = f"https://game.raceroom.com/users/{username}/career?CurrentPage=-{i}&PageSize=100&json"

      RaceHashes = []
      path_to_career_pages = user_folder / "CareerPages"
      pages = [p for p in os.listdir(path_to_career_pages) if p.endswith(".json")]
      for page in pages:
        with open(path_to_career_pages / page, "r") as f:
          raceList = json.load(f)["context"]["c"]["raceList"]["GetUserMpRatingProgressResult"]["Entries"]
          for r in raceList:
            if r["RaceHash"] not in RaceHashes:
              RaceHashes.append(r["RaceHash"])
      with ThreadPoolExecutor(max_workers=5) as executor:
        futures = {executor.submit(download_race_log, rid, user_folder): rid for rid in RaceHashes}
        for future in as_completed(futures):
          print(future.result())
